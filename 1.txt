# Вы клонируете файлы проекта с гита в консольке
git clone https://github.com/Mys2018/Arduino-based-pattern-lock.git

# Вы переключаете ветку с main на dev
git switch dev

# Вам нужно будет сделать какие-то изменения и добавить в так называемое временное хранилище этой командой
git add .

# Нужно будет закомитить изменения, т.е. их зафиксировать
git commit -m "Я -Ваше ФИ- добавил -Что-то-"

# Отправить изменения на сервер
git push
